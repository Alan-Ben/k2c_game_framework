## 数据管理

### NPPlayerSkinComponent

```csharp
public class NPPlayerSkinComponent : _ANPBasicPlayerComponent
{
    // 已解锁皮肤列表
    private Dictionary<int, SkinInfo> _m_skinDic;

    // 当前使用的皮肤
    private Dictionary<int, int> _m_curSkinDic; // type -> skinId

    // 已解锁称号列表
    private List<TitleInfo> _m_titleList;

    // 当前普通称号
    private int _m_curCommTitleId;

    // 当前组合称号
    private ComboTitleInfo _m_curComboTitle;

    /// <summary>
    /// 获取当前皮肤
    /// </summary>
    public int getCurSkinId(EPlayerSkinType type)
    {
        if (_m_curSkinDic.TryGetValue((int)type, out var skinId))
            return skinId;
        return 0; // 默认皮肤
    }

    /// <summary>
    /// 请求解锁皮肤
    /// </summary>
    public void reqUnlockSkin(int skinId, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_018_PlayerSkinOp.make_010_ReqUnlockPlayerSkin(skinId),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_018_010_RetUnlockPlayerSkin>((msg) =>
            {
                callback?.Invoke();
            }));
    }

    /// <summary>
    /// 请求设置当前皮肤
    /// </summary>
    public void reqSetCurSkin(int skinId, int skinType, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_018_PlayerSkinOp.make_011_ReqSetCurPlayerSkin(skinId, skinType),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_018_011_RetSetCurPlayerSkin>((msg) =>
            {
                callback?.Invoke();
            }));
    }

    /// <summary>
    /// 请求设置称号
    /// </summary>
    public void reqSetTitle(int titleId, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_018_PlayerSkinOp.make_001_ReqSetCommTitle(titleId),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_018_001_RetSetCommTitle>((msg) =>
            {
                callback?.Invoke();
            }));
    }
}
```

## UI 窗口

### 皮肤选择界面

```csharp
public class SkinSelectWindow : BaseWindow
{
    private EPlayerSkinType _currentType;

    public void setSkinType(EPlayerSkinType type)
    {
        _currentType = type;
        refreshSkinList();
    }

    // 刷新皮肤列表
    private void refreshSkinList()
    {
        var skinConfigList = GRefdataCoreMgr.instance.playerSkinList.getByType(_currentType);
        var skinDic = NPPlayerSkinComponent.instance.skinDic;

        foreach (var config in skinConfigList)
        {
            bool unlocked = skinDic.ContainsKey(config.skin_id);
            bool isUsing = NPPlayerSkinComponent.instance.getCurSkinId(_currentType) == config.skin_id;
            // 更新UI显示
        }
    }

    // 点击皮肤
    public void onClickSkin(int skinId)
    {
        var skinComponent = NPPlayerSkinComponent.instance;

        if (skinComponent.hasSkin(skinId))
        {
            // 已解锁，设置使用
            skinComponent.reqSetCurSkin(skinId, (int)_currentType, () =>
            {
                refreshSkinList();
            });
        }
        else
        {
            // 未解锁，请求解锁
            skinComponent.reqUnlockSkin(skinId, () =>
            {
                refreshSkinList();
            });
        }
    }
}
```

### 称号选择界面

```csharp
public class TitleSelectWindow : BaseWindow
{
    // 切换普通/组合称号
    public void switchTitleType(bool isCombo)
    {
        if (isCombo)
        {
            showComboTitleList();
        }
        else
        {
            showCommonTitleList();
        }
    }

    // 选择普通称号
    public void onSelectCommonTitle(int titleId)
    {
        NPPlayerSkinComponent.instance.reqSetTitle(titleId, () =>
        {
            refreshUI();
        });
    }

    // 选择组合称号
    public void onSelectComboTitle(int preId, int sfxId, int bgId)
    {
        NPPlayerSkinComponent.instance.reqSetComboTitle(preId, sfxId, bgId, () =>
        {
            refreshUI();
        });
    }
}
```

## 协议对接

```csharp
// 接收皮肤变化
public void onPlayerSkinChg(GS2GC_018_060_OnPlayerSkinChg msg)
{
    int skinId = msg.getSkinId();
    int skinType = msg.getSkinType();

    _m_curSkinDic[skinType] = skinId;

    WinMsg.SendMsg(WinMsgType.ON_PLAYER_SKIN_CHG, skinId, skinType);
}

// 接收称号变化
public void onCurTitleChg(GS2GC_018_054_OnCurTitleChg msg)
{
    _m_curCommTitleId = msg.getTitleId();

    WinMsg.SendMsg(WinMsgType.ON_PLAYER_TITLE_CHG, _m_curCommTitleId);
}
```

## 注意事项

1. 皮肤解锁需要消耗资源或满足条件
2. 称号有有效期限制，需要检查过期
3. 组合称号由三部分组成，需要分别解锁
