## 数据管理

### NPClientDataComponent

```csharp
public class NPClientDataComponent : _ANPBasicPlayerComponent
{
    // 数据缓存
    private Dictionary<int, Dictionary<string, ClientDataInfo>> _m_dataCache;

    // 待同步数据队列
    private Queue<ClientDataInfo> _m_pendingSyncQueue;

    // 是否有未同步数据
    private bool _m_hasUnsyncData;

    /// <summary>
    /// 获取客户端数据
    /// </summary>
    public T getData<T>(int dataType, string dataKey) where T : class
    {
        if (_m_dataCache.TryGetValue(dataType, out var typeDic))
        {
            if (typeDic.TryGetValue(dataKey, out var info))
            {
                return JsonUtility.FromJson<T>(info.dataValue);
            }
        }
        return null;
    }

    /// <summary>
    /// 设置客户端数据
    /// </summary>
    public void setData<T>(int dataType, string dataKey, T data, bool syncNow = false)
    {
        if (!_m_dataCache.ContainsKey(dataType))
        {
            _m_dataCache[dataType] = new Dictionary<string, ClientDataInfo>();
        }

        var info = new ClientDataInfo();
        info.dataType = dataType;
        info.dataKey = dataKey;
        info.dataValue = JsonUtility.ToJson(data);
        info.version = getNextVersion(dataType, dataKey);

        _m_dataCache[dataType][dataKey] = info;
        _m_hasUnsyncData = true;

        if (syncNow)
        {
            syncData(info);
        }
        else
        {
            _m_pendingSyncQueue.Enqueue(info);
        }
    }

    /// <summary>
    /// 同步数据到服务端
    /// </summary>
    private void syncData(ClientDataInfo info)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_011_ClientDataOp.make_002_ReqSaveClientData(
                info.dataType, info.dataKey, info.dataValue, info.version),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_011_002_RetSaveClientData>((msg) =>
            {
                onSyncSuccess(info, msg.getVersion());
            }));
    }

    /// <summary>
    /// 批量同步待同步数据
    /// </summary>
    public void syncPendingData()
    {
        while (_m_pendingSyncQueue.Count > 0)
        {
            var info = _m_pendingSyncQueue.Dequeue();
            syncData(info);
        }
    }
}
```

## 使用示例

### 保存设置

```csharp
// 保存音量设置
var settings = new GameSettings();
settings.bgmVolume = 0.8f;
settings.sfxVolume = 1.0f;
settings.isFullScreen = true;

NPClientDataComponent.instance.setData(
    (int)EClientDataType.Settings,
    "game_settings",
    settings,
    true);  // 立即同步
```

### 读取设置

```csharp
// 读取音量设置
var settings = NPClientDataComponent.instance.getData<GameSettings>(
    (int)EClientDataType.Settings,
    "game_settings");

if (settings != null)
{
    ApplySettings(settings);
}
```

### 引导进度管理

```csharp
public class TutorialManager
{
    // 保存引导进度
    public void saveTutorialProgress(int tutorialId, int step)
    {
        var progress = new TutorialProgress();
        progress.tutorialId = tutorialId;
        progress.currentStep = step;
        progress.completeTime = DateTime.Now.Ticks;

        NPClientDataComponent.instance.setData(
            (int)EClientDataType.Tutorial,
            $"tutorial_{tutorialId}",
            progress);
    }

    // 获取引导进度
    public int getTutorialProgress(int tutorialId)
    {
        var progress = NPClientDataComponent.instance.getData<TutorialProgress>(
            (int)EClientDataType.Tutorial,
            $"tutorial_{tutorialId}");

        return progress?.currentStep ?? 0;
    }
}
```

## 登出同步

```csharp
// 游戏退出时同步数据
public void onGameExit()
{
    NPClientDataComponent.instance.syncPendingData();
}
```

## 注意事项

1. 大数据不应存储在客户端数据中
2. 敏感数据需要加密处理
3. 注意版本兼容性，新版本需要兼容旧版本数据
4. 登出前确保数据同步完成
