### 核心功能

```csharp
public class TreasureHuntComponent : _ANPBasicPlayerComponent
{
    // 站点列表
    private List<TreasureStationInfo> _m_stationList;

    // 矿石列表
    private Dictionary<int, TreasureOreInfo> _m_oreDic;

    // 宝藏列表
    private List<TreasureInfo> _m_treasureList;

    // 请求捕获矿石
    public void reqCaptureOre(int stationId, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_036_TreasureHuntOp.make_001_ReqTreasureHuntOreCapture(stationId),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_036_001_RetTreasureHuntOreCapture>((msg) =>
            {
                callback?.Invoke();
            }));
    }

    // 请求合成
    public void reqComposite(int compositeId, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_036_TreasureHuntOp.make_006_ReqTreasureHuntCompositeActive(compositeId),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_036_006_RetTreasureHuntCompositeActive>((msg) =>
            {
                callback?.Invoke();
            }));
    }
}
```

### 核心逻辑

```java
public void captureOre(long playerId, int stationId) {
    // 1. 检查站点状态
    TreasureStation station = stationDao.selectById(stationId);
    if (station.getState() != EStationState.ACTIVE.getValue()) {
        throw new BusinessException(ErrorCode.STATION_NOT_ACTIVE);
    }

    // 2. 检查捕获力
    int capturePower = getCapturePower(playerId);
    if (capturePower <= 0) {
        throw new BusinessException(ErrorCode.NO_CAPTURE_POWER);
    }

    // 3. 执行捕获
    TreasureOreConfig oreConfig = randomOre(station);
    addOre(playerId, oreConfig.getOreId(), oreConfig.getCount());

    // 4. 消耗捕获力
    consumeCapturePower(playerId, 1);

    // 5. 推送通知
    pushOreAdd(playerId, oreConfig.getOreId(), oreConfig.getCount());
}
```

## 注意事项

1. 捕获力有上限和恢复机制
2. 不同站点产出不同类型的矿石
3. 合成需要满足材料条件
