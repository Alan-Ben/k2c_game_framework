## 核心逻辑

### 培养子女

```java
public void trainChild(long playerId, long childId, int trainType) {
    // 1. 获取子女信息
    PlayerChild child = childDao.selectById(childId);
    if (child == null || child.getPlayerId() != playerId) {
        throw new BusinessException(ErrorCode.CHILD_NOT_FOUND);
    }

    // 2. 检查状态
    if (child.getState() != EChildState.CHILD.getValue()) {
        throw new BusinessException(ErrorCode.CHILD_ALREADY_ADULT);
    }

    // 3. 检查今日培养次数
    int todayCount = getChildTodayTrainCount(playerId, childId);
    ChildTrainConfig config = trainConfigDao.getById(trainType);
    if (todayCount >= config.getDailyLimit()) {
        throw new BusinessException(ErrorCode.TRAIN_LIMIT_EXCEEDED);
    }

    // 4. 扣除消耗
    itemService.consumeItems(playerId, config.getCostList());

    // 5. 增加经验和天赋
    child.setExp(child.getExp() + config.getExpAdd());
    child.setTalent(child.getTalent() + config.getTalentAdd());

    // 6. 检查升级
    checkLevelUp(child);

    // 7. 更新数据
    childDao.updateById(child);

    // 8. 推送通知
    pushChildLvlChg(playerId, child);
}
```

### 子女毕业

```java
public void graduateChild(long playerId, long childId) {
    // 1. 获取子女信息
    PlayerChild child = childDao.selectById(childId);
    if (child == null || child.getPlayerId() != playerId) {
        throw new BusinessException(ErrorCode.CHILD_NOT_FOUND);
    }

    // 2. 检查等级
    ChildConfig config = childConfigDao.getById(child.getCfgId());
    if (child.getLevel() < config.getMaxLevel()) {
        throw new BusinessException(ErrorCode.CHILD_LEVEL_NOT_ENOUGH);
    }

    // 3. 更新状态
    child.setState(EChildState.ADULT.getValue());
    childDao.updateById(child);

    // 4. 添加到成年列表
    PlayerAdult adult = createAdultFromChild(child);
    adultDao.insert(adult);

    // 5. 推送通知
    pushChildRemove(playerId, childId);
    pushUnmarriedAdultAdd(playerId, adult);
}
```

### 申请联姻

```java
public void applyMarriage(long playerId, long targetPlayerId, long myAdultId) {
    // 1. 检查成年子女
    PlayerAdult myAdult = adultDao.selectById(myAdultId);
    if (myAdult == null || myAdult.getPlayerId() != playerId) {
        throw new BusinessException(ErrorCode.ADULT_NOT_FOUND);
    }

    if (myAdult.getMarryState() != EMarriageState.UNMARRIED.getValue()) {
        throw new BusinessException(ErrorCode.ADULT_ALREADY_MARRIED);
    }

    // 2. 检查目标玩家
    PlayerBase targetPlayer = playerDao.selectById(targetPlayerId);
    if (targetPlayer == null) {
        throw new BusinessException(ErrorCode.PLAYER_NOT_FOUND);
    }

    // 3. 检查是否拒绝联姻
    if (isRefuseMarriage(targetPlayerId)) {
        throw new BusinessException(ErrorCode.PLAYER_REFUSE_MARRIAGE);
    }

    // 4. 创建申请
    EngageRequest request = new EngageRequest();
    request.setId(generateId());
    request.setFromPlayerId(playerId);
    request.setFromAdultId(myAdultId);
    request.setToPlayerId(targetPlayerId);
    request.setStatus(ERequestStatus.PENDING.getValue());
    request.setCreateTime(new Date());
    request.setExpireTime(DateUtils.addHours(new Date(), 24));
    engageRequestDao.insert(request);

    // 5. 推送通知给目标玩家
    pushToMeApplyAdd(targetPlayerId, request);
}
```

### 同意联姻

```java
public void agreeMarriage(long playerId, long requestId) {
    // 1. 获取申请
    EngageRequest request = engageRequestDao.selectById(requestId);
    if (request == null || request.getToPlayerId() != playerId) {
        throw new BusinessException(ErrorCode.REQUEST_NOT_FOUND);
    }

    if (request.getStatus() != ERequestStatus.PENDING.getValue()) {
        throw new BusinessException(ErrorCode.REQUEST_ALREADY_HANDLED);
    }

    // 2. 检查申请方成年子女状态
    PlayerAdult fromAdult = adultDao.selectById(request.getFromAdultId());
    if (fromAdult.getMarryState() != EMarriageState.UNMARRIED.getValue()) {
        throw new BusinessException(ErrorCode.ADULT_ALREADY_MARRIED);
    }

    // 3. 更新申请状态
    request.setStatus(ERequestStatus.AGREED.getValue());
    engageRequestDao.updateById(request);

    // 4. 创建婚姻记录
    createMarriageRecord(request);

    // 5. 发放奖励
    giveMarriageReward(playerId, request.getFromPlayerId());

    // 6. 推送通知
    pushMarriageSuccess(request);
}
```

## 定时任务

### 清理过期申请

```java
@Scheduled(cron = "0 0 */1 * * ?") // 每小时执行
public void cleanExpiredRequests() {
    Date now = new Date();
    List<EngageRequest> expiredList = engageRequestDao.selectExpired(now);

    for (EngageRequest request : expiredList) {
        request.setStatus(ERequestStatus.EXPIRED.getValue());
        engageRequestDao.updateById(request);

        // 通知双方
        pushRequestExpired(request);
    }
}
```
