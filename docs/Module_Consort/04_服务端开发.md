## 核心逻辑

### 升级羁绊

```java
public void upgradeFettersLvl(long playerId, long consortId) {
    // 1. 获取伴侣信息
    PlayerConsort consort = consortDao.selectByPlayerAndId(playerId, consortId);
    if (consort == null) {
        throw new BusinessException(ErrorCode.CONSORT_NOT_FOUND);
    }

    // 2. 获取配置
    ConsortConfig config = consortConfigDao.getById(consortId);
    int nextLvl = consort.getFettersLvl() + 1;
    if (nextLvl > config.getMaxFettersLvl()) {
        throw new BusinessException(ErrorCode.FETTERS_MAX_LEVEL);
    }

    // 3. 检查亲密度是否足够
    FettersConfig fettersConfig = fettersConfigDao.getByLevel(nextLvl);
    if (consort.getIntimacy() < fettersConfig.getNeedIntimacy()) {
        throw new BusinessException(ErrorCode.INTIMACY_NOT_ENOUGH);
    }

    // 4. 升级
    consort.setFettersLvl(nextLvl);
    consortDao.updateById(consort);

    // 5. 推送通知
    pushFettersChg(playerId, consortId, nextLvl);
}
```

### 解锁皮肤

```java
public void unlockSkin(long playerId, long consortId, int skinId) {
    // 1. 检查伴侣是否存在
    PlayerConsort consort = consortDao.selectByPlayerAndId(playerId, consortId);
    if (consort == null) {
        throw new BusinessException(ErrorCode.CONSORT_NOT_FOUND);
    }

    // 2. 检查皮肤配置
    ConsortSkinConfig skinConfig = skinConfigDao.getById(skinId);
    if (skinConfig == null || skinConfig.getConsortId() != consortId) {
        throw new BusinessException(ErrorCode.SKIN_NOT_FOUND);
    }

    // 3. 检查是否已解锁
    PlayerConsortSkin existingSkin = skinDao.selectByPlayerConsortSkin(playerId, consortId, skinId);
    if (existingSkin != null) {
        throw new BusinessException(ErrorCode.SKIN_ALREADY_UNLOCKED);
    }

    // 4. 检查解锁条件
    if (!checkUnlockCondition(playerId, skinConfig.getUnlockCondition())) {
        throw new BusinessException(ErrorCode.UNLOCK_CONDITION_NOT_MET);
    }

    // 5. 扣除消耗
    if (skinConfig.getCostList() != null) {
        itemService.consumeItems(playerId, skinConfig.getCostList());
    }

    // 6. 解锁皮肤
    PlayerConsortSkin skin = new PlayerConsortSkin();
    skin.setPlayerId(playerId);
    skin.setConsortId(consortId);
    skin.setSkinId(skinId);
    skin.setUnlockTime(new Date());
    skinDao.insert(skin);

    // 7. 推送通知
    pushSkinAdd(playerId, consortId, skinId);
}
```

### AI聊天

```java
public String aiChat(long playerId, long consortId, String msg) {
    // 1. 获取伴侣信息
    PlayerConsort consort = consortDao.selectByPlayerAndId(playerId, consortId);
    if (consort == null) {
        throw new BusinessException(ErrorCode.CONSORT_NOT_FOUND);
    }

    // 2. 记录玩家消息
    saveChatRecord(playerId, consortId, EChatMsgType.PLAYER, msg);

    // 3. 调用AI服务
    ConsortConfig config = consortConfigDao.getById(consortId);
    String personality = config.getPersonality();
    String reply = aiChatService.chat(personality, msg);

    // 4. 记录伴侣回复
    saveChatRecord(playerId, consortId, EChatMsgType.CONSORT, reply);

    // 5. 推送通知
    pushAiChatMsgAdd(playerId, consortId, reply);

    return reply;
}
```

### 随机召唤

```java
public long callRandomConsort(long playerId) {
    // 1. 获取已解锁伴侣列表
    List<PlayerConsort> unlockedList = consortDao.selectUnlockedByPlayerId(playerId);

    if (unlockedList.isEmpty()) {
        throw new BusinessException(ErrorCode.NO_UNLOCKED_CONSORT);
    }

    // 2. 根据权重随机选择
    int totalWeight = 0;
    for (PlayerConsort consort : unlockedList) {
        ConsortConfig config = consortConfigDao.getById(consort.getConsortId());
        totalWeight += config.getCallWeight();
    }

    int random = ThreadLocalRandom.current().nextInt(totalWeight);
    int currentWeight = 0;
    for (PlayerConsort consort : unlockedList) {
        ConsortConfig config = consortConfigDao.getById(consort.getConsortId());
        currentWeight += config.getCallWeight();
        if (random < currentWeight) {
            pushRandCallConsortChg(playerId, consort.getConsortId());
            return consort.getConsortId();
        }
    }

    return unlockedList.get(0).getConsortId();
}
```
