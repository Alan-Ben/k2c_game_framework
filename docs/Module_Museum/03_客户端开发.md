public class MuseumComponent : _ANPBasicPlayerComponent

    {
        NPGSClientListener.sendRequest(
            NPGSWriter_035_MuseumOp.make_001_ReqMuseumItemUpgrade(itemId),

                callback?.Invoke();

            NPGSWriter_035_MuseumOp.make_002_ReqMuseumItemActive(itemId),

            {
                callback?.Invoke();
            }));
    }
}

### 核心逻辑

```java
public void upgradeItem(long playerId, int itemId) {
    PlayerMuseumItem item = museumItemDao.selectByPlayerAndId(playerId, itemId);
    if (item == null) {
        throw new BusinessException(ErrorCode.ITEM_NOT_FOUND);
    }

    MuseumItemConfig config = itemConfigDao.getById(itemId);
    if (item.getLevel() >= config.getMaxLevel()) {
        throw new BusinessException(ErrorCode.ITEM_MAX_LEVEL);
    }

    // 扣除消耗
    List<ItemInfo> cost = calculateUpgradeCost(config, item.getLevel());
    itemService.consumeItems(playerId, cost);

    // 升级
    item.setLevel(item.getLevel() + 1);
    museumItemDao.updateById(item);

    pushItemChg(playerId, itemId, item.getLevel());
}
```

## 注意事项

1. 藏品升级需要消耗材料
2. 激活藏品可以获得属性加成
3. 稀有藏品有更高的属性加成
