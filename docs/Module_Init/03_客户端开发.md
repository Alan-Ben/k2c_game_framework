# 初始化系统 - 客户端开发

## 组件结构

初始化系统客户端主要由 NPPlayerComponentMgr 负责管理所有组件的初始化流程。

## 组件基类

### _ANPBasicPlayerComponent

所有玩家组件的基类：

```csharp
public abstract class _ANPBasicPlayerComponent
{
    // 组件管理器引用
    protected NPPlayerComponentMgr _m_compMgr;

    // 是否必须初始化
    public abstract bool isMustInit { get; }

    // 组件类型
    public abstract ENPPlayerCompType compType { get; }

    // 依赖组件列表
    public abstract ENPPlayerCompType[] dependCompList { get; }

    // 是否允许提前初始化
    public virtual bool canPreInit { get { return false; } }

    // 发送初始化协议
    public virtual void presendInitProtocol() { }

    // 处理初始化
    protected abstract void _dealInit();

    // 初始化完成回调
    protected abstract void _onInitDone();

    // 初始化失败回调
    protected abstract void _onInitFail();

    // 所有组件初始化完成回调
    public virtual void onAllCompInited() { }

    // 释放资源
    protected abstract void _discard();
}
```

## 组件管理器

### NPPlayerComponentMgr

```csharp
public class NPPlayerComponentMgr
{
    // 组件字典
    private Dictionary<ENPPlayerCompType, _ANPBasicPlayerComponent> _m_compDic;

    // 初始化队列
    private Queue<ENPPlayerCompType> _m_initQueue;

    // 初始化组件
    public void initComponent(ENPPlayerCompType compType)
    {
        var comp = createComponent(compType);
        _m_compDic[compType] = comp;

        // 检查依赖
        if (!checkDependencies(comp))
        {
            _m_initQueue.Enqueue(compType);
            return;
        }

        // 开始初始化
        comp.presendInitProtocol();
    }

    // 组件初始化完成
    public void onCompInitDone(ENPPlayerCompType compType)
    {
        // 检查队列中是否有依赖此组件的
        checkPendingComponents();
    }

    // 所有组件初始化完成
    public void onAllInitDone()
    {
        foreach (var comp in _m_compDic.Values)
        {
            comp.onAllCompInited();
        }
    }
}
```

## 初始化流程

### 登录初始化

```csharp
public class LoginManager
{
    public void onLoginSuccess()
    {
        // 1. 创建组件管理器
        var compMgr = new NPPlayerComponentMgr();

        // 2. 按顺序初始化核心组件
        initCoreComponents(compMgr);

        // 3. 发送首个初始化请求
        compMgr.initComponent(ENPPlayerCompType.PLAYER_INFO);
    }

    private void initCoreComponents(NPPlayerComponentMgr compMgr)
    {
        // 核心组件列表
        var coreComps = new ENPPlayerCompType[] {
            ENPPlayerCompType.PLAYER_INFO,
            ENPPlayerCompType.CURRENCY,
            ENPPlayerCompType.BAG,
            ENPPlayerCompType.HERO,
            // ...
        };

        foreach (var compType in coreComps)
        {
            compMgr.initComponent(compType);
        }
    }
}
```

### 组件初始化示例

```csharp
public class PlayerAchieveComponent : _ANPBasicPlayerComponent
{
    public override bool isMustInit { get { return true; } }

    public override ENPPlayerCompType compType { get { return ENPPlayerCompType.ACHIEVE; } }

    public override ENPPlayerCompType[] dependCompList { get { return null; } }

    public override bool canPreInit { get { return true; } }

    public override void presendInitProtocol()
    {
        _initData();
        reqAchieveInit();
    }

    protected override void _dealInit() { }

    protected override void _onInitDone() { }

    protected override void _onInitFail()
    {
        ALLog.Error("PlayerAchieveComponent init Fail!!!");
    }

    protected override void _discard()
    {
        clear();
    }

    public override void onAllCompInited()
    {
        // 初始化成就数据上次刷新是否已满标记
        if (_m_allAchieveList != null)
        {
            for (int i = 0; i < _m_allAchieveList.Count; i++)
            {
                if(_m_allAchieveList[i] != null)
                    _m_allAchieveList[i].initIsLastFull();
            }
        }
        _refreshRedTip();
    }
}
```

### 接收初始化响应

```csharp
public void retAchieveInit(GS2GC_002_050_RetAchieveInit _msg)
{
    // 处理成就数据
    for (int i = 0; i < _msg.getAchieveList().Count; i++)
    {
        var serverInfo = _msg.getAchieveList()[i];
        var achieveInfo = getAchimentInfo(serverInfo.getAchieveId());
        if (null != achieveInfo)
        {
            achieveInfo.refreshData(serverInfo, false);
        }
    }

    // 处理成就点数据
    for (int i = 0; i < _msg.getAchievePointList().Count; i++)
    {
        var achievePointInfo = _msg.getAchievePointList()[i];
        _m_dAchievePointDic[achievePointInfo.getType()] = achievePointInfo.getCount();
        _m_dAchievePointMaxDrawStepDic[achievePointInfo.getType()] = achievePointInfo.getHadDrawMaxStep();
    }

    // 标记初始化完成
    setInitDone();
}
```

## 初始化状态管理

```csharp
public enum EInitState
{
    NONE = 0,       // 未初始化
    INITING = 1,    // 初始化中
    DONE = 2,       // 初始化完成
    FAILED = 3      // 初始化失败
}

// 设置初始化完成
protected void setInitDone()
{
    _m_initState = EInitState.DONE;
    _m_compMgr.onCompInitDone(compType);
}

// 设置初始化失败
protected void setInitFail()
{
    _m_initState = EInitState.FAILED;
    _m_compMgr.onCompInitFail(compType);
}
```

## 注意事项

1. 组件初始化顺序必须遵守依赖关系
2. 可选组件失败不应阻止进入游戏
3. 提前初始化 (canPreInit) 可以优化加载速度
4. 注意处理网络超时和重试
