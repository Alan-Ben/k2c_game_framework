## 数据管理

### PlayerConsortComponent

```csharp
public class PlayerConsortComponent : _ANPBasicPlayerComponent
{
    // 伴侣列表
    private Dictionary<long, ConsortInfo> _m_consortDic;

    // 当前激活的伴侣ID
    private long _m_curConsortId;

    /// <summary>
    /// 获取伴侣列表
    /// </summary>
    public Dictionary<long, ConsortInfo> consortDic { get { return _m_consortDic; } }

    /// <summary>
    /// 获取伴侣信息
    /// </summary>
    public ConsortInfo getConsortInfo(long consortId)
    {
        if (_m_consortDic.TryGetValue(consortId, out var info))
            return info;
        return null;
    }

    /// <summary>
    /// 请求升级羁绊
    /// </summary>
    public void reqUpgradeFettersLvl(long consortId, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_015_ConsortOp.make_001_ReqUpgradeFettersLvl(consortId),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_015_001_RetUpgradeFettersLvl>((msg) =>
            {
                callback?.Invoke();
            }));
    }

    /// <summary>
    /// 请求解锁皮肤
    /// </summary>
    public void reqUnlockSkin(long consortId, int skinId, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_015_ConsortOp.make_012_ReqUnlockSkin(consortId, skinId),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_015_012_RetUnlockSkin>((msg) =>
            {
                callback?.Invoke();
            }));
    }

    /// <summary>
    /// 请求AI聊天
    /// </summary>
    public void reqAiChat(long consortId, string msg, Action<string> callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_015_ConsortOp.make_022_ReqConsortAiChat(consortId, msg),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_015_022_RetConsortAiChat>((resp) =>
            {
                callback?.Invoke(resp.getReplyMsg());
            }));
    }
}
```

## UI 窗口

### 伴侣详情界面

```csharp
public class ConsortDetailWindow : BaseWindow
{
    private ConsortInfo _currentConsort;

    public void setConsortInfo(ConsortInfo consortInfo)
    {
        _currentConsort = consortInfo;
        refreshUI();
    }

    // 升级羁绊
    public void onClickUpgradeFetters()
    {
        PlayerConsortComponent.instance.reqUpgradeFettersLvl(_currentConsort.consortId, () =>
        {
            refreshUI();
            showUpgradeEffect();
        });
    }

    // 切换皮肤
    public void onClickChangeSkin(int skinId)
    {
        // 检查是否已解锁
        if (!_currentConsort.skinList.Contains(skinId))
        {
            // 请求解锁
            PlayerConsortComponent.instance.reqUnlockSkin(_currentConsort.consortId, skinId, () =>
            {
                refreshUI();
            });
        }
        else
        {
            // 设置当前皮肤
            PlayerConsortComponent.instance.reqSetCurSkin(_currentConsort.consortId, skinId, () =>
            {
                refreshUI();
            });
        }
    }
}
```

### 伴侣聊天界面

```csharp
public class ConsortChatWindow : BaseWindow
{
    private ConsortInfo _currentConsort;
    private List<ChatMsgInfo> _msgList;

    // 发送消息
    public void sendMsg(string content)
    {
        PlayerConsortComponent.instance.reqAiChat(_currentConsort.consortId, content, (reply) =>
        {
            // 添加玩家消息
            addPlayerMsg(content);
            // 添加伴侣回复
            addConsortReply(reply);
        });
    }

    private void addPlayerMsg(string content)
    {
        var msg = new ChatMsgInfo();
        msg.msgType = EChatMsgType.PLAYER;
        msg.content = content;
        msg.time = DateTime.Now.Ticks;
        _msgList.Add(msg);
        refreshMsgList();
    }

    private void addConsortReply(string content)
    {
        var msg = new ChatMsgInfo();
        msg.msgType = EChatMsgType.CONSORT;
        msg.content = content;
        msg.time = DateTime.Now.Ticks;
        _msgList.Add(msg);
        refreshMsgList();
    }
}
```

## 协议对接

```csharp
// 接收亲密度变化
public void onConsortIntimacyChg(GS2GC_015_051_OnConsortIntimacyChg msg)
{
    var consortId = msg.getConsortId();
    var consort = getConsortInfo(consortId);
    if (consort != null)
    {
        consort.updateIntimacy(msg.getIntimacy());
        WinMsg.SendMsg(WinMsgType.ON_CONSORT_INTIMACY_CHG, consortId);
    }
}

// 接收皮肤添加
public void onSkinAdd(GS2GC_015_055_OnSkinAdd msg)
{
    var consortId = msg.getConsortId();
    var skinId = msg.getSkinId();
    var consort = getConsortInfo(consortId);
    if (consort != null)
    {
        consort.addSkin(skinId);
        WinMsg.SendMsg(WinMsgType.ON_CONSORT_SKIN_ADD, consortId, skinId);
    }
}
```

4. 光环升级需要消耗资源
