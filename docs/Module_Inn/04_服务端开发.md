```java
/**
 * 接待客人
 * @param playerId 玩家ID
 * @param isAKey 是否一键接待
 */
public void receiveGuest(long playerId, boolean isAKey) {
    // 1. 获取玩家客栈数据
    InnComp innComp = getInnComp(playerId);

    // 2. 检查接待条件
    if (!checkCanReceive(innComp)) {
        throw new BusinessException(ErrMain.INN_NO_GUEST_TO_RECEIVE);
    }

    // 3. 获取可接待的客人和菜品
    List<InnGuestInfo> guestList = getAvailableGuests(innComp);
    List<InnDishInfo> dishList = getAvailableDishes(innComp);

    // 4. 根据随机种子分配客人和菜品
    CSSyncRandom random = new CSSyncRandom(innComp.getRandomSeed());
    int receiveCount = isAKey ? getMaxReceiveCount(innComp) : 1;

    for (int i = 0; i < receiveCount; i++) {
        long guestInstanceId = innComp.getNextGuestInstanceId();
        InnGuestInfo guest = selectGuest(random, guestInstanceId, guestList);
        InnDishInfo dish = selectDish(random, guestInstanceId, dishList);

        // 5. 添加到接待队列
        innComp.addReceiveQueue(guestInstanceId, guest, dish);
    }

    // 6. 更新接待信息
    innComp.updateReceiveList();

    // 7. 推送通知
    pushInnReceiveChg(playerId, innComp.getReceiveList());
}
```

### 2.2 结算奖励

```java
/**
 * 结算奖励
 * @param playerId 玩家ID
 * @param guestNum 结算客人数量
 * @return 结算信息
 */
public Inn_SettleInfo settle(long playerId, int guestNum) {
    // 1. 获取玩家客栈数据
    InnComp innComp = getInnComp(playerId);

    // 2. 检查可结算数量
    int availableCount = innComp.getSettleableCount();
    if (availableCount <= 0) {
        throw new BusinessException(ErrMain.INN_NO_GUEST_TO_SETTLE);
    }

    int realCount = Math.min(guestNum, availableCount);

    // 3. 创建结算信息
    Inn_SettleInfo settleInfo = new Inn_SettleInfo();
    settleInfo.setReceiveNum(realCount);

    long totalPopularity = 0;
    long totalAffection = 0;
    long totalBlueprint = 0;
    Map<Long, Inn_DishSettleInfo> dishSettleMap = new HashMap<>();

    // 4. 计算每个客人的奖励
    for (int i = 0; i < realCount; i++) {
        InnReceiveQueue queue = innComp.popReceiveQueue();

        // 计算人气值
        long popularity = calculatePopularity(queue);
        totalPopularity += popularity;

        // 计算心意值
        long affection = calculateAffection(queue);
        totalAffection += affection;

        // 随机蓝图掉落
        if (checkBlueprintDrop(queue)) {
            totalBlueprint++;
        }

        // 统计菜品熟练度
        Inn_DishSettleInfo dishSettle = dishSettleMap.computeIfAbsent(
            queue.getDishId(),
            k -> new Inn_DishSettleInfo(k, 0, 0)
        );
        dishSettle.setReceiveNum(dishSettle.getReceiveNum() + 1);

        RefInnGuest guestRef = RefInnGuest.getMgr().get(queue.getGuestId());
        long finesse = guestRef.finesse_add;
        dishSettle.setAddFinesse(dishSettle.getAddFinesse() + finesse);

        // 更新菜品熟练度
        innComp.addDishFinesse(queue.getDishId(), finesse);
    }

    // 5. 更新客栈数据
    innComp.addPopularity(totalPopularity);
    innComp.updateDishList();

    // 6. 发放奖励
    giveRewards(playerId, totalAffection, totalBlueprint);

    // 7. 设置结算信息
    settleInfo.setAddPopularity(totalPopularity);
    settleInfo.setAddAffection(totalAffection);
    settleInfo.setAddStationBlueprintNum(totalBlueprint);
    settleInfo.setDishList(new ArrayList<>(dishSettleMap.values()));

    // 8. 推送通知
    pushInnReceiveChg(playerId, innComp.getReceiveList());
    pushInnPopularityChg(playerId, innComp.getPopularity());
    pushInnDishChg(playerId, innComp.getDishList());

    // 9. 检查升级
    checkLevelUp(playerId);

    return settleInfo;
}
```

```java
/**
 * 解锁设施
 * @param playerId 玩家ID
 * @param stationId 设施ID
 */
public void unlockStation(long playerId, long stationId) {
    // 1. 获取玩家客栈数据
    InnComp innComp = getInnComp(playerId);

    // 2. 检查设施是否存在
    RefInnStation stationRef = RefInnStation.getMgr().get(stationId);
    if (stationRef == null) {
        throw new BusinessException(ErrMain.INN_STATION_NOT_FOUND);
    }

    // 3. 检查是否已解锁
    if (innComp.hasStation(stationId)) {
        throw new BusinessException(ErrMain.INN_STATION_ALREADY_BUILT);
    }

    // 4. 检查前置条件
    long needReceiveNum = stationRef.need_receive_guest_num;
    if (innComp.getHadBeenReceiveNum() < needReceiveNum) {
        throw new BusinessException(ErrMain.INN_LEVEL_NOT_ENOUGH);
    }

    // 5. 检查并扣除资源
    List<NPCommonCostItem> buildCost = stationRef.build_cost;
    if (!checkAndCost(playerId, buildCost)) {
        throw new BusinessException(ErrMain.RESOURCE_NOT_ENOUGH);
    }

    // 6. 创建设施数据
    InnStationInfo stationInfo = new InnStationInfo();
    stationInfo.setStationId(stationId);
    stationInfo.setLevel(1);
    innComp.addStation(stationInfo);

    // 7. 记录首次升级时间
    if (innComp.getFirstTimeUpgradeTimeMs() == 0) {
        innComp.setFirstTimeUpgradeTimeMs(System.currentTimeMillis());
        pushInnFirstTimeUpgradeTimeMsChg(playerId);
    }

    // 8. 推送通知
    pushInnStationAdd(playerId, stationInfo);
}
```

### 2.4 设施升级

```java
/**
 * 升级设施
 * @param playerId 玩家ID
 * @param stationId 设施ID
 */
public void upgradeStation(long playerId, long stationId) {
    // 1. 获取玩家客栈数据
    InnComp innComp = getInnComp(playerId);

    // 2. 获取设施信息
    InnStationInfo stationInfo = innComp.getStationInfo(stationId);
    if (stationInfo == null) {
        throw new BusinessException(ErrMain.INN_STATION_NOT_FOUND);
    }

    // 3. 检查等级上限
    RefInnStation stationRef = stationInfo.getRef();
    int maxLevel = stationRef.getMaxLevel();
    if (stationInfo.getLevel() >= maxLevel) {
        throw new BusinessException(ErrMain.INN_STATION_LEVEL_MAX);
    }

    // 4. 获取升级配置
    RefInnStationLevel levelRef = stationRef.getLevelRef(stationInfo.getLevel() + 1);
    if (levelRef == null) {
        throw new BusinessException(ErrMain.INN_STATION_LEVEL_MAX);
    }

    // 5. 检查并扣除资源
    if (!checkAndCost(playerId, levelRef.upgrade_cost)) {
        throw new BusinessException(ErrMain.RESOURCE_NOT_ENOUGH);
    }

    // 6. 升级设施
    stationInfo.setLevel(stationInfo.getLevel() + 1);
    innComp.updateStation(stationInfo);

    // 7. 推送通知
    pushInnStationChg(playerId, stationInfo);
}
```

### 2.5 菜品解锁

```java
/**
 * 解锁菜品
 * @param playerId 玩家ID
 * @param dishId 菜品ID
 */
public void unlockDish(long playerId, long dishId) {
    // 1. 获取玩家客栈数据
    InnComp innComp = getInnComp(playerId);

    // 2. 检查菜品是否存在
    RefInnDish dishRef = RefInnDish.getMgr().get(dishId);
    if (dishRef == null) {
        throw new BusinessException(ErrMain.INN_DISH_NOT_FOUND);
    }

    // 3. 检查是否已解锁
    InnDishInfo dishInfo = innComp.getDishInfo(dishId);
    if (dishInfo != null && dishInfo.isHadUnlock()) {
        throw new BusinessException(ErrMain.INN_DISH_ALREADY_UNLOCK);
    }

    // 4. 检查解锁条件
    if (!checkUnlockCondition(playerId, dishRef.unlock_condition)) {
        throw new BusinessException(ErrMain.INN_DISH_UNLOCK_CONDITION_NOT_MET);
    }

    // 5. 检查关联设施
    for (Long stationId : dishRef.need_station_id_list) {
        if (!innComp.hasStation(stationId)) {
            throw new BusinessException(ErrMain.INN_STATION_NOT_FOUND);
        }
    }

    // 6. 解锁菜品
    if (dishInfo == null) {
        dishInfo = new InnDishInfo();
        dishInfo.setDishId(dishId);
    }
    dishInfo.setHadUnlock(true);
    dishInfo.setLevel(1);
    innComp.addOrUpdateDish(dishInfo);

    // 7. 推送通知
    pushInnDishAdd(playerId, dishInfo);
}
```

```java
/**
 * 升级勋章
 * @param playerId 玩家ID

public void upgradeMedal(long playerId) {
    // 1. 获取玩家客栈数据
    InnComp innComp = getInnComp(playerId);

    // 2. 检查当前勋章等级
    int currentLevel = innComp.getMedalLevel();
    int nextLevel = currentLevel + 1;

    // 3. 获取下一级勋章配置
    RefInnMedalLevel nextMedalRef = RefInnMedalLevel.getMgr().get(nextLevel);
    if (nextMedalRef == null) {
        throw new BusinessException(ErrMain.INN_MEDAL_LEVEL_MAX);
    }

    // 4. 检查客栈等级要求
    if (innComp.getLevel() < nextMedalRef.need_inn_level) {
        throw new BusinessException(ErrMain.INN_LEVEL_NOT_ENOUGH);
    }

    // 5. 升级勋章
    innComp.setMedalLevel(nextLevel);

    // 6. 更新属性加成
    updateBonusModifier(playerId);

    // 7. 推送通知
    pushInnMedalLevelChg(playerId, nextLevel);
}
```

## 四、等级检查

### 4.1 检查客栈升级

```java
/**
 * 检查客栈是否可以升级
 */
protected void checkLevelUp(long playerId) {
    InnComp innComp = getInnComp(playerId);

    int currentLevel = innComp.getLevel();
    long popularity = innComp.getPopularity();

    // 获取下一级配置
    RefInnLevel nextLevelRef = RefInnLevel.getMgr().get(currentLevel + 1);
    if (nextLevelRef == null) {
        return; // 已满级
    }

    // 检查人气值是否足够
    if (popularity < nextLevelRef.need_popularity) {
        return;
    }

    // 升级
    innComp.setLevel(currentLevel + 1);
    pushInnLevelChg(playerId, currentLevel + 1);

    // 检查勋章是否可以升级
    checkMedalLevelUp(playerId);
}

/**
 * 检查勋章是否可以升级
 */
protected void checkMedalLevelUp(long playerId) {
    InnComp innComp = getInnComp(playerId);

    int currentMedalLevel = innComp.getMedalLevel();
    int innLevel = innComp.getLevel();

    // 获取下一级勋章配置
    RefInnMedalLevel nextMedalRef = RefInnMedalLevel.getMgr().get(currentMedalLevel + 1);
    if (nextMedalRef == null) {
        return; // 已满级
    }

    // 检查客栈等级
    if (innLevel < nextMedalRef.need_inn_level) {
        return;
    }

    // 自动升级勋章
    innComp.setMedalLevel(currentMedalLevel + 1);
    pushInnMedalLevelChg(playerId, currentMedalLevel + 1);

    // 递归检查是否可以继续升级
    checkMedalLevelUp(playerId);
}
```

## 六、注意事项

1. **随机种子同步**：使用同步随机种子确保客户端和服务端客人和菜品分配一致
2. **接待队列管理**：接待队列按时间顺序存储，支持批量结算
3. **属性加成**：勋章等级提供全局属性加成，升级时需要更新加成
4. **自动升级**：人气值满足条件时自动升级客栈等级
5. **推送时机**：所有数据变更后都需要推送通知给客户端
