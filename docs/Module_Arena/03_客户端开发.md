## 数据管理

### ArenaComponent

```csharp
public class ArenaComponent : _ANPBasicPlayerComponent
{
    // 竞技场信息
    private ArenaInfo _m_arenaInfo;

    // 战报列表
    private List<ArenaBattleReportInfo> _m_reportList;

    // 名人榜
    private List<ArenaCelebrityInfo> _m_celebrityList;

    /// <summary>
    /// 竞技场信息
    /// </summary>
    public ArenaInfo arenaInfo { get { return _m_arenaInfo; } }

    /// <summary>
    /// 请求随机攻击
    /// </summary>
    public void reqRandomAttack(Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_023_ArenaOp.make_001_ReqRandomAttackPlayer(),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_023_001_RetRandomAttackPlayer>((msg) =>
            {
                callback?.Invoke();
            }));
    }

    /// <summary>
    /// 请求选择英雄
    /// </summary>
    public void reqSelectHero(List<long> heroIdList, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_023_ArenaOp.make_003_ReqSelectAttackSelectHero(heroIdList),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_023_003_RetSelectAttackSelectHero>((msg) =>
            {
                callback?.Invoke();
            }));
    }

    /// <summary>
    /// 请求选择Buff
    /// </summary>
    public void reqChooseBuff(int buffId, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_023_ArenaOp.make_004_ReqChooseBuff(buffId),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_023_004_RetChooseBuff>((msg) =>
            {
                callback?.Invoke();
            }));
    }
}
```

## UI 窗口

### 竞技场主界面

```csharp
public class ArenaMainWindow : BaseWindow
{
    // 刷新竞技场信息
    public void refreshArenaInfo()
    {
        var arenaInfo = ArenaComponent.instance.arenaInfo;
        setRankText(arenaInfo.rank);
        setScoreText(arenaInfo.score);
        setWinLoseText(arenaInfo.winCount, arenaInfo.loseCount);
    }

    // 随机匹配
    public void onClickRandomMatch()
    {
        ArenaComponent.instance.reqRandomAttack(() =>
        {
            // 进入选将界面
            UIManager.openWindow<ArenaSelectHeroWindow>();
        });
    }

    // 一键攻击
    public void onClickAKeyAttack()
    {
        ArenaComponent.instance.reqAKeyAttack(() =>
        {
            refreshArenaInfo();
        });
    }

    // 查看战报
    public void onClickViewReport()
    {
        ArenaComponent.instance.reqBattleReport((reportList) =>
        {
            UIManager.openWindow<ArenaReportWindow>(reportList);
        });
    }
}
```

### 竞技场战斗界面

```csharp
public class ArenaBattleWindow : BaseWindow
{
    private List<long> _selectedHeroList;
    private int _selectedBuffId;

    // 选择英雄
    public void onSelectHero(long heroId)
    {
        if (_selectedHeroList.Contains(heroId))
        {
            _selectedHeroList.Remove(heroId);
        }
        else if (_selectedHeroList.Count < MAX_HERO_COUNT)
        {
            _selectedHeroList.Add(heroId);
        }
        refreshHeroSelection();
    }

    // 确认选择
    public void onClickConfirm()
    {
        ArenaComponent.instance.reqSelectHero(_selectedHeroList, () =>
        {
            // 进入选Buff界面
            showBuffSelection();
        });
    }

    // 选择Buff
    public void onSelectBuff(int buffId)
    {
        _selectedBuffId = buffId;
        ArenaComponent.instance.reqChooseBuff(buffId, () =>
        {
            // 开始战斗
            startBattle();
        });
    }
}
```

## 爬塔模式

### TowerComponent

```csharp
public class TowerComponent : _ANPBasicPlayerComponent
{
    // 爬塔信息
    private Dictionary<int, TowerInfo> _m_towerDic;

    /// <summary>
    /// 请求爬塔战斗
    /// </summary>
    public void reqTowerFight(int chapterId, int floor, Action<bool> callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_023_ArenaOp.make_021_ReqTowerFight(chapterId, floor),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_023_021_RetTowerFight>((msg) =>
            {
                callback?.Invoke(msg.isSuccess());
            }));
    }
}
```

## 协议对接

```csharp
// 接收竞技场信息变化
public void onArenaBaseInfoChg(GS2GC_023_052_OnArenaBaseInfoChg msg)
{
    _m_arenaInfo.updateInfo(msg.getArenaInfo());

    WinMsg.SendMsg(WinMsgType.ON_ARENA_INFO_CHG);
}

// 接收爬塔位置变化
public void onTowerPosChg(GS2GC_023_061_OnTowerPosChg msg)
{
    int chapterId = msg.getChapterId();
    int floor = msg.getFloor();

    if (_m_towerDic.TryGetValue(chapterId, out var towerInfo))
    {
        towerInfo.floor = floor;
    }

    WinMsg.SendMsg(WinMsgType.ON_TOWER_POS_CHG, chapterId, floor);
}
```

## 注意事项

2. 爬塔层数有上限
3. Buff选择影响战斗结果
