## 核心逻辑

### 招募逻辑

```java
public List<RecruitResult> recruit(long playerId, int recruitId, int count) {
    // 1. 获取招募配置
    RecruitConfig config = recruitConfigDao.getById(recruitId);
    if (config == null) {
        throw new BusinessException(ErrorCode.RECRUIT_NOT_FOUND);
    }

    // 2. 检查次数限制
    PlayerRecruit playerRecruit = recruitDao.getByPlayerAndId(playerId, recruitId);
    if (config.getDailyLimit() > 0 && playerRecruit.getTodayCount() + count > config.getDailyLimit()) {
        throw new BusinessException(ErrorCode.RECRUIT_LIMIT_EXCEEDED);
    }

    // 3. 扣除消耗
    List<ItemInfo> costList = calculateRecruitCost(config, count);
    itemService.consumeItems(playerId, costList);

    // 4. 执行招募
    List<RecruitResult> results = new ArrayList<>();
    for (int i = 0; i < count; i++) {
        RecruitResult result = doSingleRecruit(playerId, config, playerRecruit);
        results.add(result);
    }

    // 5. 更新招募数据
    playerRecruit.setTotalCount(playerRecruit.getTotalCount() + count);
    playerRecruit.setTodayCount(playerRecruit.getTodayCount() + count);
    recruitDao.updateById(playerRecruit);

    return results;
}

private RecruitResult doSingleRecruit(long playerId, RecruitConfig config, PlayerRecruit playerRecruit) {
    // 检查是否触发保底
    boolean guarantee = false;
    if (config.getGuaranteeCount() > 0 &&
        playerRecruit.getGuaranteeCount() >= config.getGuaranteeCount() - 1) {
        guarantee = true;
        playerRecruit.setGuaranteeCount(0);
    } else {
        playerRecruit.setGuaranteeCount(playerRecruit.getGuaranteeCount() + 1);
    }

    // 随机抽取
    int itemId = randomFromPool(config.getRewardPool(), guarantee);

    // 发放奖励
    itemService.giveItem(playerId, itemId, 1);

    return new RecruitResult(itemId, 1, getItemRarity(itemId));
}
```

### 抽卡逻辑

```java
public List<GachaResult> gachaRoll(long playerId, int gachaId, int count) {
    // 1. 获取抽卡配置
    GachaConfig config = gachaConfigDao.getById(gachaId);
    if (config == null || !config.isActive()) {
        throw new BusinessException(ErrorCode.GACHA_NOT_FOUND);
    }

    // 2. 检查次数限制
    if (count != 1 && count != 10) {
        throw new BusinessException(ErrorCode.INVALID_GACHA_COUNT);
    }

    // 3. 扣除消耗
    List<ItemInfo> costList = calculateGachaCost(config, count);
    itemService.consumeItems(playerId, costList);

    // 4. 执行抽卡
    PlayerGacha playerGacha = gachaDao.getByPlayerAndId(playerId, gachaId);
    List<GachaResult> results = new ArrayList<>();

    for (int i = 0; i < count; i++) {
        GachaResult result = doSingleGacha(playerId, config, playerGacha);
        results.add(result);
    }

    // 5. 更新抽卡数据
    playerGacha.setTotalCount(playerGacha.getTotalCount() + count);
    gachaDao.updateById(playerGacha);

    // 6. 推送获得物品通知
    pushGainItemList(playerId, results);

    return results;
}
```

### 红点逻辑

```java
public void refreshRedDot(long playerId, int redDotId) {
    // 获取红点配置
    RedDotConfig config = redDotConfigDao.getById(redDotId);

    // 计算红点数量
    int count = calculateRedDotCount(playerId, config);

    // 推送红点变化
    GS2GC_007_059_PushRedDotChg notification = new GS2GC_007_059_PushRedDotChg();
    notification.addRedDotList(new RedDotInfo(redDotId, count));
    sendToClient(playerId, notification);

    // 刷新父级红点
    if (config.getParentId() > 0) {
        refreshRedDot(playerId, config.getParentId());
    }
}

private int calculateRedDotCount(long playerId, RedDotConfig config) {
    switch (config.getModuleType()) {
        case ModuleType.MAIL:
            return mailService.getUnreadCount(playerId);
        case ModuleType.QUEST:
            return questService.getUnclaimedCount(playerId);
        case ModuleType.ACHIEVE:
            return achieveService.getUnclaimedCount(playerId);
        // ... 其他模块
        default:
            return 0;
    }
}
```

### 阶段目标逻辑

```java
public void takeStageGoalReward(long playerId, int goalId, int stage) {
    // 1. 获取目标配置
    StageGoalConfig config = stageGoalConfigDao.getById(goalId);
    if (config == null || config.getStage() != stage) {
        throw new BusinessException(ErrorCode.STAGE_GOAL_NOT_FOUND);
    }

    // 2. 检查完成状态
    PlayerStageGoal playerGoal = stageGoalDao.getByPlayerAndId(playerId, goalId);
    if (!isGoalCompleted(playerGoal, config)) {
        throw new BusinessException(ErrorCode.STAGE_GOAL_NOT_COMPLETED);
    }

    // 3. 检查是否已领取
    if (playerGoal.getRewardTaken().contains(stage)) {
        throw new BusinessException(ErrorCode.REWARD_ALREADY_TAKEN);
    }

    // 4. 发放奖励
    rewardService.giveRewards(playerId, config.getReward());

    // 5. 更新领取状态
    playerGoal.getRewardTaken().add(stage);
    stageGoalDao.updateById(playerGoal);

    // 6. 推送变化通知
    pushStageGoalChg(playerId, playerGoal);
}
```
