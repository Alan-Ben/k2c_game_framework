# 商店模块 - 数据配表

## 模块概述

**模块编号**: p030
**模块名称**: ShopOp (商店操作)
**主要功能**: 提供商品购买、商店刷新、礼包系统、限时礼包推送等功能

| start_time | long | 否 | 0 | 折扣开始时间（毫秒时间戳） |
| end_time | long | 否 | 0 | 折扣结束时间（毫秒时间戳） |
| vip_level | int | 否 | 0 | 所需VIP等级 |

### 折扣计算公式

```
实际价格 = 原价 × discount / 10000
```

**示例计算**:

| discount值 | 折扣 | 计算示例 |
|------------|------|----------|
| 10000 | 100%（无折扣） | 500 × 10000/10000 = 500 |
| 8000 | 80%（8折） | 500 × 8000/10000 = 400 |
| 5000 | 50%（5折） | 500 × 5000/10000 = 250 |
| 3000 | 30%（3折） | 500 × 3000/10000 = 150 |

### 折扣叠加规则

```mermaid
flowchart TD
    A[商品原价] --> B{是否有商品折扣?}
    B -->|是| C[应用商品折扣]
    B -->|否| D[保持原价]
    C --> E{是否有VIP折扣?}
    D --> E
    E -->|是| F[叠加VIP折扣]
    E -->|否| G[最终价格]
    F --> G

    subgraph 折扣限制
        H["最低折扣限制: 5折"]
        I["VIP折扣单独计算"]
    end

    G --> H
```

### 代码示例

```java
// 获取折扣配置
RefShopItemDiscount discountRef = RefShopItemDiscount.getMgr().get(discountId);
if (discountRef != null) {
    // 检查时间有效性
    long currentTime = System.currentTimeMillis();
    if (discountRef.start_time > 0 && currentTime < discountRef.start_time) {
        // 折扣未开始
        return originalPrice;
    }
    if (discountRef.end_time > 0 && currentTime > discountRef.end_time) {
        // 折扣已结束
        return originalPrice;
    }

    // 应用折扣
    long actualPrice = originalPrice * discountRef.discount / 10000;

    // 检查最低折扣限制
    long minPrice = originalPrice * 5000 / 10000; // 最低5折
    return Math.max(actualPrice, minPrice);
}
```

```java
/**
 * 商店服务实现

public class ShopService
{
    /**
     * 购买商品
     */
    public BuyResult buyShopItem(long cid, long shopRefId, long instanceId, int count)
    {
        BuyResult result = new BuyResult();

        // 1. 获取商店配置
        RefShop shopRef = RefShop.getMgr().get(shopRefId);
        if (shopRef == null) {
            result.setError(CommErr.REF_NOT_FOUND, "商店配置不存在");
            return result;
        }

        // 2. 检查商店是否解锁
        if (!checkShopUnlock(cid, shopRef)) {
            result.setError(ShopErr.SHOP_NOT_UNLOCKED, "商店未解锁");
            return result;
        }

        // 3. 获取商店数据
        ShopBO shopBO = shopDao.getByPlayerAndRef(cid, shopRefId);
        if (shopBO == null) {
            shopBO = initPlayerShop(cid, shopRefId);
        }

        // 4. 查找商品
        ShopItemBO item = shopBO.findItem(instanceId);
        if (item == null) {
            result.setError(ShopErr.ITEM_NOT_FOUND, "商品不存在");
            return result;
        }

        // 5. 检查购买次数
        RefShopItem itemRef = RefShopItem.getMgr().get(item.refId);
        if (itemRef.buy_limit > 0 && item.buyCount + count > itemRef.buy_limit) {
            result.setError(ShopErr.BUY_LIMIT_REACHED, "购买次数已达上限");
            return result;
        }

        // 6. 计算价格
        long unitPrice = calculateItemPrice(item, itemRef);
        long totalCost = unitPrice * count;

        // 7. 检查货币
        if (!currencyService.hasEnough(cid, itemRef.cost_item.item, totalCost)) {
            result.setError(CommErr.CURRENCY_NOT_ENOUGH, "货币不足");
            return result;
        }

        // 8. 检查背包
        List<RewardItem> rewards = buildRewardList(itemRef, count);
        if (!bagService.hasSpace(cid, rewards)) {
            result.setError(CommErr.BAG_FULL, "背包已满");
            return result;
        }

        // 9. 执行购买事务
        try {
            // 扣除货币
            currencyService.cost(cid, itemRef.cost_item.item, totalCost, "商店购买");

            // 发放商品
            bagService.addItems(cid, rewards, "商店购买");

            // 更新购买记录
            item.buyCount += count;
            shopDao.updateShopItem(cid, item);

            // 记录日志
            logPurchase(cid, shopRefId, item, count, totalCost);

            result.setSuccess(rewards, totalCost);
        } catch (Exception e) {
            log.error("购买商品失败", e);
            result.setError(CommErr.INTERNAL_ERROR, "购买失败");
        }

        return result;
    }

    /**
     * 刷新商店
     */
    public RefreshResult refreshShop(long cid, long shopRefId, boolean isFree)
    {
        RefreshResult result = new RefreshResult();

        // 1. 获取商店配置
        RefShop shopRef = RefShop.getMgr().get(shopRefId);
        if (shopRef == null) {
            result.setError(CommErr.REF_NOT_FOUND, "商店配置不存在");
            return result;
        }

        // 2. 获取商店数据
        ShopBO shopBO = shopDao.getByPlayerAndRef(cid, shopRefId);
        if (shopBO == null) {
            shopBO = initPlayerShop(cid, shopRefId);
        }

        // 3. 检查刷新权限
        if (isFree) {
            int freeCount = getFreeRefreshCount(cid, shopRef);
            if (freeCount <= 0) {
                result.setError(ShopErr.FREE_REFRESH_USED, "免费刷新次数已用完");
                return result;
            }
        } else {
            // 检查付费刷新限制
            if (shopRef.pay_refresh_limit > 0 && shopBO.refreshNum >= shopRef.pay_refresh_limit) {
                result.setError(ShopErr.REFRESH_LIMIT_REACHED, "刷新次数已达上限");
                return result;
            }

            // 计算刷新费用
            long cost = calculateRefreshCost(shopRef, shopBO.refreshNum);
            if (!currencyService.hasEnough(cid, ECurrency.DIAMOND, cost)) {
                result.setError(CommErr.CURRENCY_NOT_ENOUGH, "货币不足");
                return result;
            }

            // 扣除费用
            currencyService.cost(cid, ECurrency.DIAMOND, cost, "商店刷新");
            result.setCost(cost);
        }

        // 4. 生成新商品
        List<ShopItemBO> newItems = generateShopItems(shopRef);
        shopBO.goodsList = newItems;
        shopBO.refreshNum++;
        shopBO.nextRefreshTime = calculateNextRefreshTime(shopRef);

        // 5. 保存数据
        shopDao.update(shopBO);

        result.setShopInfo(shopBO);
        return result;
    }

    /**
     * 生成商店商品
     */
    private List<ShopItemBO> generateShopItems(RefShop shopRef)
    {
        List<ShopItemBO> items = new ArrayList<>();

        // 获取商品池
        List<RefShopItem> itemPool = RefShopItem.getMgr().getByShopId(shopRef.id);

        // 构建权重选择器
        WeightedSelector<RefShopItem> selector = new WeightedSelector<>();
        for (RefShopItem itemRef : itemPool) {
            selector.add(itemRef, itemRef.weight);
        }

        // 随机选择商品
        for (int i = 0; i < shopRef.slot_count; i++) {
            RefShopItem selected = selector.select();
            ShopItemBO item = createShopItem(selected);
            items.add(item);
        }

        return items;
    }

    /**
     * 计算商品价格（含折扣）
     */
    private long calculateItemPrice(ShopItemBO item, RefShopItem itemRef)
    {
        long basePrice = itemRef.cost_item.count;

        // 应用折扣
        if (item.discountRefId > 0) {
            RefShopItemDiscount discount = RefShopItemDiscount.getMgr().get(item.discountRefId);
            if (discount != null) {
                basePrice = basePrice * discount.discount / 10000;
            }
        }

        // 最低5折保护
        long minPrice = itemRef.cost_item.count / 2;
        return Math.max(basePrice, minPrice);
    }
}
```

```java
public List<Shop_Info> batchGetShopInfo(long cid, List<Long> shopRefIds)

    // 先从缓存批量获取
    Map<Long, ShopBO> cachedShops = cacheManager.batchGet(shopRefIds);

    List<Long> uncachedIds = shopRefIds.stream()
        .filter(id -> !cachedShops.containsKey(id))

    // 批量查询数据库
    if (!uncachedIds.isEmpty()) {

    return shopRefIds.stream()
        .map(id -> cachedShops.get(id))
        .filter(Objects::nonNull)
        .map(ShopBO::toShopInfo)
        .collect(Collectors.toList());
}
```

```java
public class ShopLogQueue
{
    private static final BlockingQueue<ShopLogRecord> queue =
        new LinkedBlockingQueue<>(10000);

    private static final ExecutorService executor =
        Executors.newSingleThreadExecutor();

    static {
        executor.submit(() -> {
            while (true) {
                List<ShopLogRecord> batch = new ArrayList<>();
                ShopLogRecord first = queue.poll(100, TimeUnit.MILLISECONDS);
                if (first != null) {
                    batch.add(first);
                    queue.drainTo(batch, 100);
                    shopLogDao.batchInsert(batch);
                }
            }
        });
    }

    public static void addLog(ShopLogRecord record)
    {
        if (!queue.offer(record)) {
            log.warn("日志队列已满，丢弃日志");
        }
    }
```

```java
        for (RefShopItem itemRef : RefShopItem.getMgr().getAll()) {
            errors.addAll(validateShopItem(itemRef));
        }

        // 校验关联关系
        errors.addAll(validateRelations());

        return errors;
    }

    /**
     * 校验单个商店配置
     */
    private List<ConfigError> validateShop(RefShop shopRef)
    {
        List<ConfigError> errors = new ArrayList<>();

        // 检查必填字段
        if (shopRef.id <= 0) {
            errors.add(new ConfigError("shop", shopRef.id, "id必须大于0"));
        }
        if (shopRef.name == null || shopRef.name.isEmpty()) {
            errors.add(new ConfigError("shop", shopRef.id, "name不能为空"));
        }
        if (shopRef.type <= 0 || shopRef.type > 10) {
            errors.add(new ConfigError("shop", shopRef.id, "type必须在1-10范围内"));
        }

        // 检查刷新配置
        if (shopRef.refresh_type == ENPShopRefreshType.DAILY.ordinal()) {
            if (shopRef.auto_refresh_time_list == null || shopRef.auto_refresh_time_list.isEmpty()) {
                errors.add(new ConfigError("shop", shopRef.id, "每日刷新必须配置刷新时间点"));
            }
        }

        // 检查槽位数量
        if (shopRef.slot_count <= 0 || shopRef.slot_count > 20) {
            errors.add(new ConfigError("shop", shopRef.id, "slot_count必须在1-20范围内"));
        }

        return errors;
    }

    /**
     * 校验关联关系
     */
    private List<ConfigError> validateRelations()
    {
        List<ConfigError> errors = new ArrayList<>();

        // 校验商品的商店关联
        for (RefShopItem itemRef : RefShopItem.getMgr().getAll()) {
            RefShop shopRef = RefShop.getMgr().get(itemRef.shop_id);
            if (shopRef == null) {
                errors.add(new ConfigError("shop_item", itemRef.id,
                    "关联的商店不存在: " + itemRef.shop_id));
            }

            // 校验商品组关联
            if (itemRef.group_id > 0) {
                RefShopItemGroup groupRef = RefShopItemGroup.getMgr().get(itemRef.group_id);
                if (groupRef == null) {
                    errors.add(new ConfigError("shop_item", itemRef.id,
                        "关联的商品组不存在: " + itemRef.group_id));
                }
            }

            // 校验折扣关联
            if (itemRef.discount_id > 0) {
                RefShopItemDiscount discountRef = RefShopItemDiscount.getMgr().get(itemRef.discount_id);
                if (discountRef == null) {
                    errors.add(new ConfigError("shop_item", itemRef.id,
                        "关联的折扣配置不存在: " + itemRef.discount_id));
                }
            }
        }

        return errors;
    }
}
```

### 热更新实现代码

```java
/**
 * 商店配置热更新处理器
 */
public class ShopConfigHotUpdater
{
    /**
     * 处理配置更新
     */
    public void onConfigUpdate(ConfigUpdateEvent event)
    {
        String configType = event.getConfigType();
        List<Long> changedIds = event.getChangedIds();

        switch (configType) {
            case "shop":
                updateShopConfig(changedIds);
                break;
            case "shop_item":
                updateShopItemConfig(changedIds);
                break;
            case "shop_item_discount":
                updateShopItemDiscountConfig(changedIds);
                break;
            case "gift_pack":
                updateGiftPackConfig(changedIds);
                break;
        }

        // 清除相关缓存
        clearRelatedCache(configType, changedIds);

        // 记录更新日志
        logConfigUpdate(event);
    }

    /**
     * 更新商店配置
     */
    private void updateShopConfig(List<Long> changedIds)
    {
        for (Long shopId : changedIds) {
            // 重新加载配置
            RefShop newRef = loadShopConfig(shopId);
            RefShop.getMgr().update(shopId, newRef);

            // 推送更新通知给在线玩家
            pushShopConfigUpdate(shopId);
        }
    }

    /**
     * 清除相关缓存
     */
    private void clearRelatedCache(String configType, List<Long> changedIds)
    {
        for (Long id : changedIds) {
            // 清除Redis缓存
            String pattern = "shop:*:" + id;
            RedisClient.deleteByPattern(pattern);

            // 清除本地缓存
            LocalCacheManager.invalidate(configType, id);
        }
    }
}
```

```java
public class ShopConfigBackupService

    private static final String BACKUP_PATH = "/data/config_backup/shop/";

    public void restoreBackup(String version)

        restoreConfigFile("shop.json", backupDir);
        restoreConfigFile("shop_item.json", backupDir);

        restoreConfigFile("shop_item_discount.json", backupDir);
        restoreConfigFile("gift_pack.json", backupDir);
        restoreConfigFile("limit_gift_pack.json", backupDir);

        // 重新加载配置
        reloadAllConfigs();

        // 清除缓存
        clearAllCache();
    }
```

```java
/**
 * 配置查询优化示例
 */
public class ShopConfigQueryOptimized
{
    // 预构建索引
    private Map<Integer, List<RefShop>> shopTypeIndex;
    private Map<Long, List<RefShopItem>> shopItemIndex;
    private Map<Integer, List<RefLimitGiftPack>> giftPackTriggerIndex;

    /**
     * 初始化索引
     */
    public void initIndexes()
    {
        // 构建商店类型索引
        shopTypeIndex = RefShop.getMgr().getAll()
            .stream()
            .collect(Collectors.groupingBy(shop -> shop.type));

        // 构建商品所属商店索引
        shopItemIndex = RefShopItem.getMgr().getAll()
            .stream()
            .collect(Collectors.groupingBy(item -> item.shop_id));

        // 构建礼包触发类型索引
        giftPackTriggerIndex = RefLimitGiftPack.getMgr().getAll()
            .stream()
            .collect(Collectors.groupingBy(gift -> gift.trigger_type));
    }

    /**
     * 快速查询：按商店类型获取商品列表
     */
    public List<RefShopItem> getShopItemsByShopType(int shopType)
    {
        List<RefShop> shops = shopTypeIndex.getOrDefault(shopType, Collections.emptyList());
        List<RefShopItem> items = new ArrayList<>();
        for (RefShop shop : shops) {
            items.addAll(shopItemIndex.getOrDefault(shop.id, Collections.emptyList()));
        }
        return items;
    }
}
```
