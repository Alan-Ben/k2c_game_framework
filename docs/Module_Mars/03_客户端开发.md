```csharp
public class MarsComponent : _ANPBasicPlayerComponent
{
    // 火星信息
    private MarsInfo _m_marsInfo;

    // 当前阶段
    private int _m_currentStage;

    // 请求前往火星
    public void reqStartToGoMars(Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_038_MarsOp.make_001_ReqStartToGoMars(),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_038_001_RetStartToGoMars>((msg) =>
            {
                callback?.Invoke();
            }));
    }

    // 请求到达阶段
    public void reqArriveStage(int stage, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_038_MarsOp.make_002_ReqArriveMarsStage(stage),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_038_002_RetArriveMarsStage>((msg) =>
            {
                callback?.Invoke();
            }));
    }
}
```

### 核心逻辑

```java
public void startToGoMars(long playerId) {
    // 1. 检查是否已在火星
    PlayerMarsRoute route = marsRouteDao.selectById(playerId);
    if (route != null && route.getRouteState() == EMarsRouteState.TRAVELING.getValue()) {
        throw new BusinessException(ErrorCode.ALREADY_ON_MARS);
    }

    // 2. 创建航程
    if (route == null) {
        route = new PlayerMarsRoute();
        route.setPlayerId(playerId);
    }
    route.setCurrentStage(0);
    route.setRouteState(EMarsRouteState.TRAVELING.getValue());
    marsRouteDao.saveOrUpdate(route);

    // 3. 推送通知
    pushMarsStart(playerId);
}
```

## 注意事项

1. 火星航程有时间限制
2. 建筑升级需要消耗资源
3. 人员分配影响探索效率
