## 数据管理

### PlayerAchieveComponent

```csharp
public class PlayerAchieveComponent : _ANPBasicPlayerComponent
{
    // 成就列表
    private List<AchieveInfo> _m_allAchieveList;

    // 成就点数字典
    private Dictionary<long, long> _m_dAchievePointDic;

    // 已领取最大阶段
    private Dictionary<long, long> _m_dAchievePointMaxDrawStepDic;

    public override bool isMustInit { get { return true; } }
    public override ENPPlayerCompType compType { get { return ENPPlayerCompType.ACHIEVE; } }
    public override bool canPreInit { get { return true; } }

    /// <summary>
    /// 根据类型获取成就列表
    /// </summary>
    public List<AchieveInfo> getInfosByType(EAchieveType type)
    {
        return _m_allAchieveList.FindAll(a => a.achieveType == type);
    }

    /// <summary>
    /// 获取成就点数
    /// </summary>
    public long getAchievePoint(EAchieveType type)
    {
        return _m_dAchievePointDic.TryGetValue((long)type, out var value) ? value : 0;
    }

    /// <summary>
    /// 请求领取成就奖励
    /// </summary>
    public void reqDoneAchieveStep(long achieveId, int step, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_021_PlayerInfoOp.make_025_ReqDoneAchieveStep(achieveId, step),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_021_025_RetDoneAchieveStep>((msg) =>
            {
                callback?.Invoke();
            }));
    }
}
```

## 协议对接

```csharp
// 接收初始化数据
public void retAchieveInit(GS2GC_002_050_RetAchieveInit msg)
{
    foreach (var serverInfo in msg.getAchieveList())
    {
        var achieveInfo = getAchimentInfo(serverInfo.getAchieveId());
        if (achieveInfo != null)
        {
            achieveInfo.refreshData(serverInfo, false);
        }
    }

    foreach (var pointInfo in msg.getAchievePointList())
    {
        _m_dAchievePointDic[pointInfo.getType()] = pointInfo.getCount();
        _m_dAchievePointMaxDrawStepDic[pointInfo.getType()] = pointInfo.getHadDrawMaxStep();
    }

    setInitDone();
}

// 接收成就变化
public void onAchieveChg(GS2GC_021_066_OnAchieveChg msg)
{
    var serverInfo = msg.getAchieve();
    var achieveInfo = getAchimentInfo(serverInfo.getAchieveId());
    if (achieveInfo != null)
    {
        achieveInfo.refreshData(serverInfo);
        _refreshRedTip();
        WinMsg.SendMsg(WinMsgType.ON_ACHIEVE_INFO_CHG, achieveInfo);
    }
}
```
