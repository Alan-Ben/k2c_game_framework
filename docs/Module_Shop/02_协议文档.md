### 2.1 GC2GS_030_001_ReqBuyShopItem - 购买商品

**主序号**: 30 | **子序号**: 1

#### 协议字段

| 字段名 | 类型 | 说明 | 必填 | 校验规则 |
|--------|------|------|------|----------|
| shopRefId | long | 商店配置ID | 是 | 必须为有效商店ID |
| instanceId | long | 商品实例ID | 是 | 必须存在于商店商品列表 |
| count | long | 购买数量 | 是 | 范围 1-999 |

#### 协议源码

```csharp
public class GC2GS_030_001_ReqBuyShopItem : _IALProtocolStructure
{
    private long shopRefId;   // 商店配置id
    private long instanceId;  // 商品id
    private long count;       // 数量

    public byte getMainOrder() { return (byte)30; }
    public byte getSubOrder() { return (byte)1; }
}
```

#### 请求示例

```csharp
// 通过商店组件发送购买请求
NPPlayer.instance.shopComp.reqBuyShopItem(shopId, instanceId, count, () => {
    // 购买成功回调
    Debug.Log("购买成功");
});
```

```csharp
// 监听器处理
public class GSSubDealer_030_050_OnShopRefresh : _AGSSubDealerBase
{
    public override void deal(NPPlayer _player, GS2GC_030_050_OnShopRefresh _msg)
    {

        _player.shopComp.retOnShopRefresh(shopInfo);

// 组件处理
public void retOnShopRefresh(Shop_Info _info)
{
    if (null == _info) return;

    NPPlayerShop shop = getShop(_info.getShopRefId());
    if (null == shop) return;

    shop.update(_info);
    setShopNextRefreshTimeMs(_info.getShopRefId(), _info.getNextRefreshTimeMs());
    _m_redDealer?.refrshRedTip(shop);
}
```

#### 客户端处理

```csharp
// 监听器处理
public class GSSubDealer_030_051_OnShopItemChg : _AGSSubDealerBase
{
    public override void deal(NPPlayer _player, GS2GC_030_051_OnShopItemChg _msg)
    {
        long shopId = _msg.getShopRefId();
        Shop_ItemInfo itemInfo = _msg.getShopItem();
        _player.shopComp.retOnShopItemChg(shopId, itemInfo);
    }
}

// 组件处理
public void retOnShopItemChg(long _shopId, Shop_ItemInfo _item)
{
    if (null == _item) return;

    NPPlayerShop shop = getShop(_shopId);
    if (null == shop) return;

    shop.updateItem(_item);
}

// 商品更新
public void updateItem(Shop_ItemInfo _item)
{
    NPPlayerShopItem temp = null;
    for (int i = 0; i < _m_shopItemList.Count; i++)
    {
        temp = _m_shopItemList[i];
        if (temp.instanceId == _item.getInstanceId())
        {
            temp.update(_item);
            break;
        }
    }
}
```

## 五、协议交互流程

### 5.1 商品购买完整流程

```mermaid
sequenceDiagram
    participant C as 客户端
    participant S as 服务端
    participant DB as 数据库
    participant Bag as 背包模块
    participant Currency as 货币模块

    C->>S: GC2GS_030_001_ReqBuyShopItem<br/>(shopRefId, instanceId, count)

    S->>S: 验证商店是否解锁
    S->>S: 验证商品是否存在
    S->>S: 验证购买次数限制

    S->>Currency: 检查货币是否足够
    Currency-->>S: 返回货币数量

    alt 货币不足
        S-->>C: 返回错误码: 货币不足
    else 货币足够
        S->>Currency: 扣除货币
        S->>Bag: 发放物品
        S->>DB: 更新购买记录
        S->>DB: 记录购买日志

        S-->>C: GS2GC_030_001_RetBuyShopItem<br/>(result=0, rewards)

        alt 达到购买上限
            S->>S: 标记商品售罄
            S-->>C: GS2GC_030_051_OnShopItemChg<br/>(shopRefId, itemInfo)
        end
    end
```

### 5.2 商店刷新完整流程

```mermaid
sequenceDiagram
    participant C as 客户端
    participant S as 服务端
    participant DB as 数据库
    participant Currency as 货币模块
    participant Random as 随机服务

    C->>S: GC2GS_030_003_ReqRefreshShop<br/>(shopRefId)

    S->>S: 检查刷新次数限制
    S->>S: 计算刷新费用

    alt 免费刷新
        S->>S: 检查免费次数
    else 付费刷新
        S->>Currency: 检查货币
        S->>Currency: 扣除货币
    end

    S->>Random: 随机生成新商品列表
    Random-->>S: 返回商品列表

    S->>DB: 更新商店数据
    S->>DB: 记录刷新日志

    S-->>C: GS2GC_030_050_OnShopRefresh<br/>(shopInfo)
```

### 5.3 自动刷新定时流程

```mermaid
sequenceDiagram
    participant Timer as 定时器
    participant S as 服务端
    participant DB as 数据库
    participant C as 客户端

    loop 每分钟检查
        Timer->>S: 检查需要刷新的商店
        S->>DB: 查询next_refresh_time <= 当前时间
        DB-->>S: 返回商店列表

        loop 遍历每个商店
            S->>S: 随机生成新商品
            S->>DB: 更新商店数据
            S-->>C: GS2GC_030_050_OnShopRefresh
        end
    end
```

### 5.5 限时礼包触发流程

```mermaid
sequenceDiagram
    participant Game as 游戏事件
    participant S as 服务端
    participant DB as 数据库
    participant C as 客户端

    Game->>S: 触发事件(升级/充值等)
```

```csharp
/// <summary>
/// 商店购买管理器
/// </summary>
public class ShopBuyManager
{
    /// <summary>
    /// 购买商品
    /// </summary>
    public void BuyItem(long shopId, long instanceId, int count)
    {
        // 1. 获取商品信息
        NPPlayerShop shop = NPPlayer.instance.shopComp.getShop(shopId);
        if (shop == null)
        {
            UIMessageBox.Show("商店不存在");
            return;
        }

        // 2. 查找商品
        List<NPPlayerShopItem> items = new List<NPPlayerShopItem>();
        shop.getShopItemList(items);
        NPPlayerShopItem item = items.FirstOrDefault(i => i.instanceId == instanceId);

        if (item == null)
        {
            UIMessageBox.Show("商品不存在");
            return;
        }

        // 3. 检查购买次数
        if (item.buyNumMax > 0 && item.lastBuyCount < count)
        {
            UIMessageBox.Show("购买次数超限");
            return;
        }

        // 4. 获取价格
        NPCommonCostItem costItem = item.getCostItem();
        long totalCost = costItem.count * count;

        // 5. 检查货币
```

```java
/**
 * 批量协议发送器
 */
public class BatchProtocolSender
{
    private static final int BATCH_SIZE = 100;
    private static final int BATCH_TIMEOUT_MS = 50;

    private Queue<ProtocolTask> taskQueue = new ConcurrentLinkedQueue<>();
    private ScheduledExecutorService scheduler;

    /**
     * 添加协议任务
     */
    public void addTask(long cid, Object protocol)
    {
        taskQueue.offer(new ProtocolTask(cid, protocol));

        if (taskQueue.size() >= BATCH_SIZE) {
            flushTasks();
        }
    }

    /**
     * 批量发送
     */
    private void flushTasks()
    {
        List<ProtocolTask> batch = new ArrayList<>();
        ProtocolTask task;
        while (batch.size() < BATCH_SIZE && (task = taskQueue.poll()) != null) {
            batch.add(task);
        }

        if (!batch.isEmpty()) {
            // 按cid分组发送
            Map<Long, List<Object>> groupedProtocols = batch.stream()
                .collect(Collectors.groupingBy(t -> t.cid,
                    Collectors.mapping(t -> t.protocol, Collectors.toList())));

            for (Map.Entry<Long, List<Object>> entry : groupedProtocols.entrySet()) {
                sendBatchProtocols(entry.getKey(), entry.getValue());
            }
        }
    }
}
```

```java
public class ReplayAttackPreventer
{
    private static final int NONCE_EXPIRE_SECONDS = 60;
    private static final String NONCE_KEY_PREFIX = "protocol:nonce:";

    /**
     * 检查并记录Nonce
     */
    public boolean checkAndRecordNonce(long cid, long timestamp, String nonce)
    {
        // 检查时间戳有效性
        long currentTime = System.currentTimeMillis();
        if (Math.abs(currentTime - timestamp) > NONCE_EXPIRE_SECONDS * 1000) {
            return false;
        }

        // 检查Nonce是否已使用
        String key = NONCE_KEY_PREFIX + cid + ":" + nonce;
        if (RedisClient.exists(key)) {
            return false; // 重放攻击
        }

        // 记录Nonce
        RedisClient.setex(key, NONCE_EXPIRE_SECONDS, "1");
        return true;
    }

    /**
     * 生成Nonce
     */
    public String generateNonce()
    {
        return UUID.randomUUID().toString().replace("-", "");
    }
}
```

```python
商店协议压力测试脚本

import asyncio
import time
from protocol_client import ProtocolClient

class ShopProtocolStressTest:
    def __init__(self, server_address, concurrent_users=100):
        self.server_address = server_address
        self.concurrent_users = concurrent_users
        self.results = []

    async def buy_item_test(self, user_id):
        """购买商品测试"""
        client = ProtocolClient(self.server_address)
        await client.connect()

        start_time = time.time()
        response = await client.send_buy_request(
            shop_id=1001,
            instance_id=generate_instance_id(),
            count=1
        )
        end_time = time.time()

        self.results.append({
            'user_id': user_id,
            'duration_ms': (end_time - start_time) * 1000,
            'success': response.result == 0
        })

        await client.disconnect()

    async def run_test(self, total_requests=1000):
        """运行压力测试"""
        tasks = []
        for i in range(total_requests):
            user_id = i % self.concurrent_users
            tasks.append(self.buy_item_test(user_id))

        start_time = time.time()
        await asyncio.gather(*tasks)
        total_time = time.time() - start_time

        # 分析结果
        success_count = sum(1 for r in self.results if r['success'])
        avg_duration = sum(r['duration_ms'] for r in self.results) / len(self.results)
        qps = total_requests / total_time

        print(f"总请求数: {total_requests}")
        print(f"成功数: {success_count}")
        print(f"成功率: {success_count / total_requests * 100:.2f}%")
        print(f"平均响应时间: {avg_duration:.2f}ms")
        print(f"QPS: {qps:.2f}")
```

```java
public class ProtocolDocGenerator
{
    /**
     * 生成协议文档
     */
    public String generateProtocolDoc(Class<?> protocolClass)
    {
        StringBuilder doc = new StringBuilder();

        // 获取协议注解
        ProtocolDesc desc = protocolClass.getAnnotation(ProtocolDesc.class);
        if (desc != null) {
            doc.append("### ").append(desc.name()).append("\n\n");
            doc.append("**说明**: ").append(desc.description()).append("\n\n");
        }

        // 获取字段列表
        doc.append("#### 协议字段\n\n");
        doc.append("| 字段名 | 类型 | 说明 | 必填 |\n");
        doc.append("|--------|------|------|------|\n");

        for (Field field : protocolClass.getDeclaredFields()) {
            FieldDesc fieldDesc = field.getAnnotation(FieldDesc.class);
            if (fieldDesc != null) {
                doc.append("| ").append(field.getName()).append(" ");
                doc.append("| ").append(field.getType().getSimpleName()).append(" ");
                doc.append("| ").append(fieldDesc.description()).append(" ");
                doc.append("| ").append(fieldDesc.required() ? "是" : "否").append(" |\n");
            }
        }

        return doc.toString();
    }
}
```
