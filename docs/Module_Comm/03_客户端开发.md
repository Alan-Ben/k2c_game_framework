## 招募组件

### RecruitComponent

```csharp
public class RecruitComponent : _ANPBasicPlayerComponent
{
    // 招募商店列表
    private Dictionary<int, RecruitShopInfo> _m_shopDic;

    // 招募记录
    private List<RecruitRecordInfo> _m_recordList;

    /// <summary>
    /// 获取招募商店信息
    /// </summary>
    public RecruitShopInfo getShopInfo(int shopId)
    {
        if (_m_shopDic.TryGetValue(shopId, out var info))
            return info;
        return null;
    }

    /// <summary>
    /// 请求招募
    /// </summary>
    public void reqRecruit(int recruitId, int count, Action<List<RecruitResult>> callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_007_CommOp.make_028_ReqRecruit(recruitId, count),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_007_028_RetRecruit>((msg) =>
            {
                callback?.Invoke(msg.getResultList());
            }));
    }
}
```

## 红点系统

### RedTipMgr

红点管理器，管理所有模块的红点状态：

```csharp
public class RedTipMgr : Singleton<RedTipMgr>
{
    // 红点数据字典
    private Dictionary<int, RedDotInfo> _m_redDotDic;

    /// <summary>
    /// 设置红点数量
    /// </summary>
    public void setCountByRefRedTipId(int redTipId, long count)
    {
        if (!_m_redDotDic.TryGetValue(redTipId, out var info))
        {
            info = new RedDotInfo(redTipId);
            _m_redDotDic[redTipId] = info;
        }

        info.count = count;

        // 发送红点变化消息
        WinMsg.SendMsg(WinMsgType.ON_RED_DOT_CHG, redTipId, count);
    }

    /// <summary>
    /// 获取红点数量
    /// </summary>
    public long getCountByRedTipId(int redTipId)
    {
        if (_m_redDotDic.TryGetValue(redTipId, out var info))
            return info.count;
        return 0;
    }

    /// <summary>
    /// 清除红点
    /// </summary>
    public void clearRedDot(int redTipId)
    {
        setCountByRefRedTipId(redTipId, 0);
    }
}
```

### 红点界面使用

```csharp
public class RedDotWidget : MonoBehaviour
{
    [SerializeField] private int _redTipId;
    [SerializeField] private GameObject _redDotObj;
    [SerializeField] private Text _countText;

    private void OnEnable()
    {
        WinMsg.Register(WinMsgType.ON_RED_DOT_CHG, onRedDotChg);
        refreshRedDot();
    }

    private void OnDisable()
    {
        WinMsg.Unregister(WinMsgType.ON_RED_DOT_CHG, onRedDotChg);
    }

    private void onRedDotChg(object[] args)
    {
        int redTipId = (int)args[0];
        if (redTipId == _redTipId)
        {
            refreshRedDot();
        }
    }

    private void refreshRedDot()
    {
        long count = RedTipMgr.instance.getCountByRedTipId(_redTipId);
        _redDotObj.SetActive(count > 0);
        _countText.text = count > 99 ? "99+" : count.ToString();
    }
}
```

## 抽卡界面

```csharp
public class GachaWindow : BaseWindow
{
    private int _currentGachaId;

    // 单抽
    public void onClickSingleRoll()
    {
        CommComponent.instance.reqGachaRoll(_currentGachaId, 1, (resultList) =>
        {
            showGachaResult(resultList);
        });
    }

    // 十连
    public void onClickTenRoll()
    {
        CommComponent.instance.reqGachaRoll(_currentGachaId, 10, (resultList) =>
        {
            showGachaResult(resultList);
        });
    }

    // 显示抽卡结果
    private void showGachaResult(List<GachaResult> resultList)
    {
        // 播放动画
        PlayGachaAnimation(resultList, () =>
        {
            // 显示结果界面
            UIManager.openWindow<WindowGachaResult>(resultList);
        });
    }
}
```

## 协议对接

```csharp
public class CommComponent : _ANPBasicPlayerComponent
{
    // 请求抽卡
    public void reqGachaRoll(int gachaId, int count, Action<List<GachaResult>> callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_007_CommOp.make_026_ReqGachaRoll(gachaId, count),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_007_026_RetGachaRoll>((msg) =>
            {
                callback?.Invoke(msg.getResultList());
            }));
    }

    // 接收红点变化
    public void onRedDotChg(GS2GC_007_059_PushRedDotChg msg)
    {
        foreach (var redDotInfo in msg.getRedDotList())
        {
            RedTipMgr.instance.setCountByRefRedTipId(
                redDotInfo.getRedDotId(),
                redDotInfo.getCount());
        }
    }

    // 接收红点移除
    public void onRedDotRemove(GS2GC_007_060_PushRedDotRemove msg)
    {
        foreach (var redDotId in msg.getRedDotIdList())
        {
            RedTipMgr.instance.clearRedDot(redDotId);
        }
    }
}
```

## 注意事项

1. 红点系统需要支持层级关系，父红点数量为子红点之和
2. 抽卡结果需要播放动画
3. 招募保底计数需要正确显示
4. 阶段目标需要检查完成条件
