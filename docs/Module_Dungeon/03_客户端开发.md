## 数据管理

### MiddayDungeonComponent

```csharp
public class MiddayDungeonComponent : _ANPBasicPlayerComponent
{
    // 副本信息
    private MiddayDungeonInfo _m_dungeonInfo;

    // 宝箱列表
    private List<MiddayDungeonBoxInfo> _m_boxList;

    /// <summary>
    /// 请求攻击
    /// </summary>
    public void reqAttack(Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_024_DungeonOp.make_001_ReqMiddayDungeonAttack(),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_024_001_RetMiddayDungeonAttack>((msg) =>
            {
                callback?.Invoke();
            }));
    }

    /// <summary>
    /// 请求领取宝箱
    /// </summary>
    public void reqDrawBox(int boxId, Action callback)
    {
        NPGSClientListener.sendRequest(
            NPGSWriter_024_DungeonOp.make_004_ReqMiddayDungeonDrawBox(boxId),
            new CommonErrCodeRequestCallbackProtocolDealer<GS2GC_024_004_RetMiddayDungeonDrawBox>((msg) =>
            {
                callback?.Invoke();
            }));
    }
}
```

### EveningDungeonComponent

```csharp
public class EveningDungeonComponent : _ANPBasicPlayerComponent

}
```

## UI 窗口

### 午间副本界面

```csharp
public class MiddayDungeonWindow : BaseWindow
{
    // 攻击
    public void onClickAttack()
    {
        MiddayDungeonComponent.instance.reqAttack(() =>
        {
            refreshUI();
            showAttackEffect();
        });
    }

    // 领取宝箱
    public void onClickDrawBox(int boxId)
    {
        MiddayDungeonComponent.instance.reqDrawBox(boxId, () =>
        {
            refreshBoxList();
            showRewardEffect();
        });
    }
}
```

### 晚间副本界面

```csharp
public class EveningDungeonWindow : BaseWindow
{
    private EveningDungeonBossInfo _bossInfo;

    // 攻击Boss
    public void onClickAttackBoss()
    {
        EveningDungeonComponent.instance.reqAttackBoss((damage) =>
        {
            showDamageEffect(damage);
            refreshBossHP();
        });
    }

    // 查看排行榜
    public void onClickViewRank()
    {
        EveningDungeonComponent.instance.reqRankList((rankList) =>
        {
            UIManager.openWindow<EveningDungeonRankWindow>(rankList);
        });
    }
}
```

## 协议对接

```csharp
// 接收午间副本信息变化
public void onMiddayDungeonInfoChg(GS2GC_024_051_OnMiddayDungeonInfoChg msg)
{
    _m_dungeonInfo.updateInfo(msg.getDungeonInfo());

    WinMsg.SendMsg(WinMsgType.ON_MIDDAY_DUNGEON_CHG);
}

// 接收晚间副本信息变化
public void onEveningDungeonInfoChg(GS2GC_024_061_OnEveningDungeonInfoChg msg)
{
    _m_bossInfo.updateInfo(msg.getBossInfo());

    WinMsg.SendMsg(WinMsgType.ON_EVENING_DUNGEON_CHG);
}

// 接收公会副本推送
public void onDungeonPush(GS2GC_037_056_OnDungeonPush msg)
{
    // 处理公会副本推送
    handleDungeonPush(msg);

    WinMsg.SendMsg(WinMsgType.ON_GUILD_DUNGEON_PUSH);
}
```

## 注意事项

1. 午间副本有开启时间限制
2. 晚间副本Boss血量全服共享
3. 公会副本需要公会解锁
